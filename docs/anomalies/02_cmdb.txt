WITH m AS (
    SELECT DISTINCT ltrim(regexp_replace(inv_no, '\D', '', 'g'), '0') AS k
    FROM stage.cmdb_movements_parsed
),
a AS (
    SELECT DISTINCT ltrim(regexp_replace(inv_no, '\D', '', 'g'), '0') AS k
    FROM stage.cmdb_assets_parsed

    (SELECT count(*) FROM m) AS mov_keys,
